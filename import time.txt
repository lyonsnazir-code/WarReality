import time
import os
import requests
from dotenv import load_dotenv
from logger import record_and_notify_trade

load_dotenv()
GAMMA_URL = "https://gamma-api.polymarket.com/markets"

# Use a persistent session for high-speed TCP connection pooling
session = requests.Session()

# Compliance / Risk Guardrails for Profit Stability
MAX_POSITION_USDC = 10.0
MIN_24H_VOLUME = 500_000.0  # Filters high-liquidity markets to prevent slippage

def fetch_compliant_markets():
    try:
        response = session.get(
            GAMMA_URL,
            params={
                "active": "true", 
                "closed": "false", 
                "limit": 10, 
                "order": "volume24hr", 
                "ascending": "false"
            },
            timeout=5
        )
        response.raise_for_status()
        return response.json()
    except Exception as e:
        print(f"[API ERROR] {e}")
        return []

def run_live_architecture():
    print("[INFO] High-Speed Compliant Architecture Initialized...")
    print(f"[RISK CONFIG] Max Position: ${MAX_POSITION_USDC} USDC | Min Volume: ${MIN_24H_VOLUME:,.2f}")
    
    while True:
        try:
            markets = fetch_compliant_markets()
            
            for m in markets:
                question = m.get("question", "Unknown Market")
                condition_id = m.get("conditionId", "0x0")
                volume = float(m.get("volume24hr", 0) or 0)
                
                # Compliance & Stability Filter: Skip low liquidity noise
                if volume < MIN_24H_VOLUME:
                    continue
                
                print(f"[QUALIFIED] {question[:45]}... | Vol: ${volume:,.2f}")
                
                # ==========================================
                # LIVE EXECUTION TRIGGER HOOK:

                # record_and_notify_trade(
                #     market_id=condition_id,

                #     entry_price=0.45,

                # )
            
            time.sleep(5)
            
        except KeyboardInterrupt:
            print("\n[INFO] Safe shutdown complete.")
            break

            print(f"[ERROR] {e}")
            time.sleep(3)

    run_live_architecture()
